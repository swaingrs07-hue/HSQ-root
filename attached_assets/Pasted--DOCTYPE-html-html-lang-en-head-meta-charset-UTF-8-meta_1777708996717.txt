  <script>
    // Animated percentage counter
    const percentCounter = document.getElementById('percent-counter');
    let currentPercent = 0;
    const targetPercent = 100;
    const duration = 5000; // Duration of loading
    const startTime = 1200; // Delay before counter starts

    function animatePercent() {
      const now = Date.now();
      const elapsed = now - animationStart;
      
      if (elapsed < startTime) {
        setTimeout(animatePercent, 16);
        return;
      }

      const animationElapsed = elapsed - startTime;
      const progress = Math.min(animationElapsed / duration, 1);
      
      // Easing function for smooth progression
      const easeProgress = progress < 0.5 
        ? 2 * progress * progress 
        : -1 + (4 - 2 * progress) * progress;
      
      currentPercent = Math.floor(easeProgress * targetPercent);
      percentCounter.textContent = currentPercent + '%';

      if (progress < 1) {
        requestAnimationFrame(animatePercent);
      }
    }

    const animationStart = Date.now();
    animatePercent();
  </script>
