🧮 Core Logic Example
function calculatePrice(basePrice, occupancy, isWeekend) {
  let price = basePrice;

  if (occupancy > 0.8) price *= 1.2;
  else if (occupancy < 0.4) price *= 0.85;

  if (isWeekend) price *= 1.1;

  return Math.round(price);
}
