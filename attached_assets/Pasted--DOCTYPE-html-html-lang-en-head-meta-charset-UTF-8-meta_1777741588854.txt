    <script>
        // Parallax Effect for Hero
        window.addEventListener('scroll', () => {
            const scrolled = window.pageYOffset;
            const heroBg = document.querySelector('.parallax-bg');
            const heroContent = document.querySelector('header .relative.z-10');
            
            if (heroBg) {
                heroBg.style.transform = `translateY(${scrolled * 0.4}px) scale(${1 + scrolled * 0.0005})`;
            }
            if (heroContent) {
                heroContent.style.opacity = 1 - (scrolled / 700);
                heroContent.style.transform = `translateY(${scrolled * 0.2}px)`;
            }

            // Reveal animations
            const reveals = document.querySelectorAll('[data-scroll]');
            reveals.forEach(el => {
                const rect = el.getBoundingClientRect();
                if (rect.top < window.innerHeight * 0.85) {
                    el.classList.add('visible');
                }
            });

            // Navbar background transition
            const nav = document.getElementById('main-nav');
            if (scrolled > 50) {
                nav.style.padding = '1rem 2rem';
                nav.classList.add('glass-nav');
            } else {
                nav.style.padding = '1.5rem 2rem';
                nav.style.backgroundColor = 'transparent';
                nav.style.borderBottomColor = 'transparent';
            }
        });

        // Initial Trigger
        window.dispatchEvent(new Event('scroll'));

        // Letter split for line reveal
        document.addEventListener('DOMContentLoaded', () => {
            const headers = document.querySelectorAll('.line-reveal');
            headers.forEach(h => h.classList.add('visible'));
        });
    </script>
