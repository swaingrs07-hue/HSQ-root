  <script>
    const hallway = document.getElementById('hallway');

      { vol: '04', side: 'right', z: -2100, title: 'Celestial Noir', desc: 'Deep contrast studies in cosmic light.', date: 'Winter 2023', loc: 'New York, US', img: 'https://images.unsplash.com/photo-1444703686981-a3abbc4d4fe3?auto=format&fit=crop&q=80&w=800', img2: 'https://images.unsplash.com/photo-1534067783941-51c9c23ecefd?auto=format&fit=crop&q=80&w=800', img3: 'https://images.unsplash.com/photo-1492144534655-ae79c964c9d7?auto=format&fit=crop&q=80&w=800' },
      { vol: '05', side: 'left', z: -2800, title: 'Obsidian Flow', desc: 'The fluid nature of dark materials.', date: 'Fall 2023', loc: 'Paris, FR', img: 'https://images.unsplash.com/photo-1486406146926-c627a92ad1ab?auto=format&fit=crop&q=80&w=800', img2: 'https://images.unsplash.com/photo-1554118811-1e0d58224f24?auto=format&fit=crop&q=80&w=800', img3: 'https://images.unsplash.com/photo-1517502884422-41eaadeff171?auto=format&fit=crop&q=80&w=800' }
    ];

    function createHallway() {
      FRAMES_DATA.forEach((data) => {
        const frame = document.createElement('div');
        frame.className = `art-frame art-frame-${data.side}`;
        const rot = data.side === 'left' ? 35 : -35;
        frame.style.transform = `translate3d(0, 0, ${data.z}px) rotateY(${rot}deg)`;
        
        frame.innerHTML = `
          <div class="w-full h-full relative group pointer-events-none">
            <img src="${data.img}" class="w-full h-full object-cover opacity-80 group-hover:opacity-100 transition-opacity duration-500">
            <div class="absolute inset-0 bg-gradient-to-t from-black/80 to-transparent flex flex-col justify-end p-6">
              <p class="text-[10px] tracking-widest uppercase text-white/40 mb-1 font-semibold">Vol. ${data.vol}</p>
              <h3 class="text-sm font-serif text-[#D4AF37] transform translate-y-2 opacity-0 group-hover:translate-y-0 group-hover:opacity-100 transition-all">View Details</h3>
            </div>
          </div>
        `;

        frame.addEventListener('click', () => openModal(data));
        hallway.appendChild(frame);
      });
    }

    function createParticles() {
      for (let i = 0; i < 40; i++) {
        const p = document.createElement('div');
        p.className = 'particle';
        p.style.left = Math.random() * 100 + '%';
        p.style.top = Math.random() * 100 + '%';
        p.style.opacity = Math.random() * 0.15;
        p.style.transform = `scale(${Math.random() * 2})`;
        particlesContainer.appendChild(p);
      }
    }

    function updateParallax() {
      const scrolled = window.scrollY;
      const translateZ = scrolled * 0.7;
      
      const frames = document.querySelectorAll('.art-frame');
      frames.forEach((frame, i) => {
        const data = FRAMES_DATA[i];
        const currentZ = data.z + translateZ;
        const distance = Math.abs(currentZ);
        const rot = data.side === 'left' ? 35 : -35;
        
        frame.style.transform = `translate3d(0, 0, ${currentZ}px) rotateY(${rot}deg)`;
        
        const blur = Math.max(0, (distance - 500) / 300);
        const brightness = Math.max(0.6, 1 - distance / 3000);
        frame.style.filter = `blur(${blur}px) brightness(${brightness})`;
      });
    }

    function openModal(data) {
      document.getElementById('modal-title').textContent = data.title;
      document.getElementById('modal-desc').textContent = data.desc;
      document.getElementById('modal-date').textContent = data.date;
      document.getElementById('modal-location').textContent = data.loc;
      
      document.getElementById('gallery-img-1').src = data.img;
      document.getElementById('gallery-img-2').src = data.img2;
      document.getElementById('gallery-img-3').src = data.img3;
      
      modalOverlay.classList.add('active');
      document.body.style.overflow = 'hidden';
    }

    function closeModal() {
      modalOverlay.classList.remove('active');
      document.body.style.overflow = 'auto';
    }

    modalOverlay.addEventListener('click', (e) => {
      if (e.target === modalOverlay) closeModal();
    });

    document.addEventListener('DOMContentLoaded', () => {
      createHallway();
      createParticles();
      window.addEventListener('scroll', updateParallax);
      updateParallax();
    });
  </script>
