// DOM Elements
const hallway = document.getElementById('hallway');
const modalOverlay = document.getElementById('modal-overlay');
const particlesContainer = document.getElementById('ambient-particles');

// Initialize
function init() {
  createArtFrames();
  createParticles();
  setupScrollListener();
  setupModalListener();
}

// Create Art Frames
function createArtFrames() {
  FRAMES_CONFIG.forEach(frame => {
    const frameEl = document.createElement('div');
    frameEl.className = `art-frame art-frame-${frame.side}`;
    frameEl.style.transform = `translate3d(0, 0, ${frame.zPosition}px) rotateY(${frame.side === 'left' ? '35deg' : '-35deg'})`;
    frameEl.onclick = () => openModal(frame);

    const img = document.createElement('img');
    img.src = frame.image;
    img.alt = frame.title;

    const label = document.createElement('div');
    label.className = 'frame-label';
    label.textContent = `Vol. ${frame.id}`;

    frameEl.appendChild(img);
    frameEl.appendChild(label);
    hallway.appendChild(frameEl);
  });
}

// Create Ambient Particles
function createParticles() {
  for (let i = 0; i < 50; i++) {
    const p = document.createElement('div');
    p.className = 'particle';
    p.style.left = Math.random() * 100 + '%';
    p.style.top = Math.random() * 100 + '%';
    p.style.opacity = Math.random() * 0.15;
    p.style.transform = `scale(${Math.random() * 2})`;
    particlesContainer.appendChild(p);
  }
}

// Scroll Listener - 3D Parallax
function setupScrollListener() {
  window.addEventListener('scroll', () => {
    const scrolled = window.scrollY;
    const translateZ = scrolled * 0.7;

    // Move hallway
    hallway.style.transform = `translateZ(${translateZ}px)`;

    // Apply depth-of-field effect
    const frames = document.querySelectorAll('.art-frame');
    frames.forEach(frame => {
      const transform = frame.style.transform;
      const zMatch = transform.match(/translate3d\(0, 0, ([-0-9.]+)px\)/);

      if (zMatch) {
        const frameZ = parseFloat(zMatch[1]);
        const distance = Math.abs(translateZ + frameZ);
        const blur = Math.max(0, (distance - 300) / 200);
        const brightness = Math.max(0.4, 1 - distance / 2000);

        frame.style.filter = `blur(${blur}px) brightness(${brightness})`;
      }
    });
  });
}

// Modal - Open
function openModal(frameData) {
  document.getElementById('modal-title').textContent = frameData.title;
  document.getElementById('modal-desc').textContent = frameData.description;
  document.getElementById('modal-date').textContent = frameData.date;
  document.getElementById('modal-location').textContent = frameData.location;

  document.getElementById('gallery-img-1').src = frameData.gallery[0];
  document.getElementById('gallery-img-2').src = frameData.gallery[1];
  document.getElementById('gallery-img-3').src = frameData.gallery[2];

  modalOverlay.classList.remove('hidden');
  document.body.style.overflow = 'hidden';
}

// Modal - Close
function closeModal() {
  modalOverlay.classList.add('hidden');
  document.body.style.overflow = 'auto';
}

// Modal - Click Outside
function setupModalListener() {
  modalOverlay.addEventListener('click', (e) => {
    if (e.target === modalOverlay) closeModal();
  });
}

// Start
document.addEventListener('DOMContentLoaded', init);
